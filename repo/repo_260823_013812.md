# FILE: aggregate_hvt.jq ::
########################################################################
. as $records

| [
    $records[]
    | select(
        .record_type == "lot"
        and .lock_state == "10.HVT"
      )
  ]

| group_by(.symbol)

| map(
    sort_by(.open_datetime_raw)
    | . as $lots
    | $lots[0] as $oldest

    | $oldest

    | .record_type = "aggregate"
    | .lock_state  = "15.HVT"

    | .quantity =
        ($lots | map(.quantity) | add)

    | .open_price =
        ($lots | map(.open_price) | max)

    | .cost_basis_price =
        ($lots | map(.cost_basis_price) | max)

    | .fifo_pnl_unrealized =
        (
            ($lots | map(.fifo_pnl_unrealized) | add)
            - 1.12
        )
  )

| ($records + .)[]

      $lots;

      if .[$i].record_type == "lot"
      and .[$i].lock_state == "10.HVT"

      and

      (
        any(
          .[$i+1:][];

          .record_type == "lot"
          and .lock_state == "50.LOS"

      )

      then

        .[$i].lock_state = "30.LKD"

# FILE: classify_locking.jq ::
########################################################################
group_by(.symbol)

| map(

    . as $lots

    |

    reduce range(0; length) as $i (

      $lots;

      if .[$i].record_type == "lot"
      and .[$i].lock_state == "30.LKD"

      then

        (
          .[$i+1:]
          | map(
              select(
                .record_type == "lot"
                and .lock_state == "50.LOS"
              )
            )
          | last
        ) as $locker

        |

        if $locker then

          .[$i].locking_lot_id = $locker.lot_id

          |
          .[$i].lock_condition =
            "◆"
            + ((($locker.open_price + 3 / $locker.quantity) | floor) | tostring)
            + "x"
            + ($locker.quantity | tostring)
            + "▶"
            + ($locker.open_date | sub("^20"; ""))

        else

          .

        end

      else

        .

      end

    )

)

| add
| .[]

# FILE: classify_pba.jq ::
########################################################################

group_by(.symbol)

| map(

    (
      map(
        select(
          .record_type == "lot"
          and .lock_state == "50.LOS"
        )
      )
    ) as $los

    |

    if ($los | length) == 0
    then
      .
    else

      (
        $los
        | max_by(.cost_basis_price)
        | .cost_basis_price
      ) as $max_cost

      |

      map(

        if .record_type == "lot"
        and .lock_state == "50.LOS"
        and .cost_basis_price == $max_cost
        and .cost_basis_price >= .mark_price

        then

          .lock_state = "20.PBA"

        else

          .

        end

      )

    end

)

| add
| .[]

# FILE: consolidate_lots.jq ::
########################################################################

group_by([
  .symbol,
  .open_datetime_raw,
  .originating_transaction_id,
  .originating_order_id
])

| map(

    if length == 1 then .[0]

    else

      .[0] + {

        quantity:
          (map(.quantity) | add),

        position_value:
          (map(.position_value) | add),

        fifo_pnl_unrealized:
          (map(.fifo_pnl_unrealized) | add),

        cost_basis_price:
          (
            (map(.cost_basis_price * .quantity) | add)
            /
            (map(.quantity) | add)
          ),

        open_price:
          (
            (map(.open_price * .quantity) | add)
            /
            (map(.quantity) | add)
          )

      }

    end

)

| .[]

#-----------------------------------------------------------------------
# update_mark_prices.jq
#
# Input:
#   One JSONL stream containing:
#     - Flex records (lots + summaries)
#     - Live records (record_type == "live")
#
# Output:
#   Updated Flex records only.
#-----------------------------------------------------------------------

# Build an index of live records.
reduce inputs as $r
(
  {
    "live": {},
    "flex": []
  };

  if $r.record_type == "live" then
    .live[$r.symbol] = $r
  else
    .flex += [$r]
  end
)

# Emit the updated Flex records.
| . as $db
| $db.flex[]
| if $db.live[.symbol] then
    .mark_price = $db.live[.symbol].mark_price
    | .position_value =
        (
            .mark_price
            * .quantity
            * .multiplier
        )
    | .fifo_pnl_unrealized =
        (
            .position_value
            - (
                .cost_basis_price
                * .quantity
                * .multiplier
            )
        )
  else
    .
  end

define prefix_filename =
${strip $()
	${eval
		path   := ${strip $(1)}
		prefix := ${strip $(2)}
	}
	${eval
		filename  := ${notdir $(path)}
	}
	$(dir_tidata)/$(prefix)$(filename:jq=jsonl)
}
endef

#-----------------------------------------------------------------------
# NOTE — PRIVATE MACRO CONVENTION:
# A leading underscore identifies a private/internal Make macro.
# Private macros are implementation helpers and are not part of the
# module's public macro interface. They may be called by public macros
# within the module but should not be treated as externally callable
# interfaces.
#
# This convention is intentional and should be considered part of the
# project's Make macro naming standard. Future directive audits should
# preserve this convention and, if appropriate, formalize it explicitly
# in the directive.
##......................................................................
define _jq_arg_validation =
${eval 
	options  := ${strip $(1)}
	script   := ${strip $(2)}
	inputs   := ${strip $(3)}
	prefix   := ${strip $(4)}
}
${eval
	input1 := ${word 1, $(inputs)}
}
${eval
	fl_out  = ${dir $(input1)}$(prefix)${notdir $(input1)}
}
: echo -n                            \
	"${notdir $(input1)}"              \
	"--> ${notdir $(script)}"          \
	"--> ${notdir $(fl_out)}"          \
;
endef
##......................................................................
define _jq_transform =
	jq $(options) --from-file 
	$(script)
	$(inputs) 
	> 
	$(input1).tmp;
endef
##......................................................................
define _jq_publish =
	mv --force 
	$(input1).tmp 
	$(input1);
	jq .
	$(input1) 
	> 
	$(fl_out);
endef
##......................................................................
define jq_stage  =
${call _jq_arg_validation, $(1), $(2), $(3), $(4)}
${strip $()
	${call _jq_transform}
	${call _jq_publish}
	:
}
endef

#-----------------------------------------------------------------------
define execute_on_rule =
${strip 
${if ${findstring ${strip $(1)}, $(MAKECMDGOALS)}, ${strip $(2)}}
}
endef

	{                                                    \
		find . -type f                                     \
		| grep --invert-match                              \
		--regexp=/@/                                       \
		--regexp=ver                                       \
		--regexp=/data/                                    \
		--regexp=/repo/                                    \
		--regexp=/notes/                                   \
		--regexp=/sh/clean_output.sh                       \
		--regexp=/sh/refactor_identifiers.*                \

		| grep                                             \
		--regexp=/data/.*_sample.*                         \
		--regexp=/notes/errors.txt                         \
		;                                                  \

	${call redirect_on_rule, dfmf, > file.manifest}      \

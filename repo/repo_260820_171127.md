#-----------------------------------------------------------------------
# update_mark_prices.jq
#
# Input:
#   One JSONL stream containing:
#     - Flex records (lots + summaries)
#     - Live records (record_type == "live")
#
# Output:
#   Updated Flex records only.
#-----------------------------------------------------------------------

# Build an index of live records.
reduce inputs as $r
(
  {
    "live": {},
    "flex": []
  };

  if $r.record_type == "live" then
    .live[$r.symbol] = $r
  else
    .flex += [$r]
  end
)

# Emit the updated Flex records.
| . as $db
| $db.flex[]
| if $db.live[.symbol] then
    .mark_price = $db.live[.symbol].mark_price
    | .position_value =
        (
            .mark_price
            * .quantity
            * .multiplier
        )
    | .fifo_pnl_unrealized =
        (
            .position_value
            - (
                .cost_basis_price
                * .quantity
                * .multiplier
            )
        )
  else
    .
  end

define prefix_filename =
${strip $()
	${eval
		path   := ${strip $(1)}
		prefix := ${strip $(2)}
	}
	${eval
		filename  := ${notdir $(path)}
	}
	$(dir_tidata)/$(prefix)$(filename:jq=jsonl)
}
endef

#-----------------------------------------------------------------------
# NOTE — PRIVATE MACRO CONVENTION:
# A leading underscore identifies a private/internal Make macro.
# Private macros are implementation helpers and are not part of the
# module's public macro interface. They may be called by public macros
# within the module but should not be treated as externally callable
# interfaces.
#
# This convention is intentional and should be considered part of the
# project's Make macro naming standard. Future directive audits should
# preserve this convention and, if appropriate, formalize it explicitly
# in the directive.
##......................................................................
define _jq_arg_validation =
${eval 
	options  := ${strip $(1)}
	script   := ${strip $(2)}
	inputs   := ${strip $(3)}
	prefix   := ${strip $(4)}
}
${eval
	input1 := ${word 1, $(inputs)}
}
${eval
	fl_out  = ${dir $(input1)}$(prefix)${notdir $(input1)}
}
: echo -n                            \
	"${notdir $(input1)}"              \
	"--> ${notdir $(script)}"          \
	"--> ${notdir $(fl_out)}"          \
;
endef
##......................................................................
define _jq_transform =
	jq $(options) --from-file 
	$(script)
	$(inputs) 
	> 
	$(input1).tmp;
endef
##......................................................................
define _jq_publish =
	mv --force 
	$(input1).tmp 
	$(input1);
	jq .
	$(input1) 
	> 
	$(fl_out);
endef
##......................................................................
define jq_stage  =
${call _jq_arg_validation, $(1), $(2), $(3), $(4)}
${strip $()
	${call _jq_transform}
	${call _jq_publish}
	:
}
endef

def wait_for_market_data(ib, tickers, timeout=10):
    start = time.time()

    while time.time() - start < timeout:
        if all(
            not math.isnan(t.markPrice)
            or not math.isnan(t.last)
            for t in tickers
        ):
            return True

        ib.sleep(0.20)

    return False

def get_historical_last_bar(ib: IB, contract):
    """Get last completed 1-min bar."""
    try:
        bars = ib.reqHistoricalData(
            contract,
            endDateTime='',
            durationStr='1 D',
            barSizeSetting='1 min',
            # barSizeSetting='10 secs',
            whatToShow='TRADES',
            useRTH=True,
            formatDate=1
        )
        if not bars:
            return None
        last = bars[-1]
        return last.low, last.high, (last.high + last.low) / 2, last.close
    except Exception as e:
        logging.warning(f"Historical data failed for {contract.symbol}: {e}")
        return None

def create_synthetic_ticker(contract,
                            account,
                            avg_cost,
                            bid,
                            ask,
                            mark,
                            last):
    """
    Build a ticker-like object from historical data.
    """

    class SyntheticTicker:
        def __init__(self):
            self.contract = contract
            self.account = account
            self.avgCost = avg_cost

            # Match the interface of ib_insync.Ticker
            self.marketDataType = 2
            self.time = None
            self.markPrice = mark

            self._bid = bid
            self._ask = ask
            self._last = last

        def marketPrice(self):
            return self.markPrice

        @property
        def bid(self):
            return self._bid

        @property
        def ask(self):
            return self._ask

        @property
        def last(self):
            return self._last

    return SyntheticTicker()

def is_market_open(ib):
    """
    Return True if the current IBKR server time falls within
    regular U.S. market hours (09:30–16:00 ET, Mon–Fri).
    """

    # return True  #******************** TESTING  ********************
    return False #******************** TESTING  ********************

    try:
        now = ib.reqCurrentTime()

        # Weekend
        if now.weekday() >= 5:
            return False

        minutes = now.hour * 60 + now.minute

        market_open = 4 * 60 + 30
        market_close = 20 * 60

        return market_open <= minutes < market_close

    except Exception as e:
        logging.warning(
            f"Failed to get server time: {e}. Assuming market closed."
        )
        return False

def render_jsonl(tickers, fp=sys.stdout):

    """
    for t in tickers:

        rec = {
            "account"  : t.account,
            "symbol"   : t.contract.symbol,
            "exchange" : t.contract.exchange,

            "mark" : None if math.isnan(t.markPrice) else t.markPrice,
            "bid"  : None if math.isnan(t.bid)       else t.bid,
            "ask"  : None if math.isnan(t.ask)       else t.ask,
            "last" : None if math.isnan(t.last)      else t.last,

            "time" : None if t.time is None else str(t.time),
            "market_data_type" : t.marketDataType
        }

        print(json.dumps(rec, separators=(',', ':')), file=fp)

    else:
        tickers = get_historical_prices(ib, positions)

    return tickers

function trim_quotes(s)
{
    gsub(/^"/, "", s)
    gsub(/"$/, "", s)
    gsub(/""/, "\"", s)
    return s
}

function json_escape(s)
{
    gsub(/\\/,"\\\\",s)
    gsub(/"/,"\\\"",s)
    gsub(/\t/,"\\t",s)
    gsub(/\r/,"\\r",s)
    gsub(/\n/,"\\n",s)
    return s
}

function split_dt(dt, arr,   d,t)
{
    arr["date"] = ""
    arr["time"] = ""

    if (dt == "")
        return

    split(dt, tmp, ";")

    d = tmp[1]
    t = tmp[2]

    if (length(d) == 8)
        arr["date"] = substr(d,1,4) "-" substr(d,5,2) "-" substr(d,7,2)

    if (length(t) == 6)
        arr["time"] = substr(t,1,2) ":" substr(t,3,2) ":" substr(t,5,2)
}

# NOTE — GENERIC SCRIPT DISPATCH:
# run_script_on_source() is intentionally a generic service dispatcher.
# Its responsibility is to select the appropriate execution engine from
# the source script's extension (AWK, SQL, or jq).
#
# Keep this generic mechanism in the shared shell library; pipeline-specific
# processing belongs in the corresponding pipeline module.
function run_script_on_source()
{
  local fl_script=${1};
  local fl_source=${2};

  local dir_script=${fl_script%/*};
  local ext_script=${fl_script##*.};
  local prompt=$(date +'%m/%d/%H:%M:%S-%a$');

  # echo -e sqlite3 -table ${fl_source} '\\\n<' ${fl_script};
  # sqlite3 -table ${fl_source} < ${fl_script};
  if [ ${ext_script} == awk ]; then {
    echo -e "${prompt} awk -f ${fl_script} ${fl_source}";
    awk -f ${fl_script} ${fl_source};
  }
  elif [ ${ext_script} == sql ]; then {
    echo -e "sqlite3 -table ${fl_source} \n< ${fl_script}";
    sqlite3 -table ${fl_db} < ${fl_sql};
  }
  elif [ ${ext_script} == jq ]; then {
    echo -e "${prompt} jq --raw-output --slurp --from-file ${fl_script} ${fl_source}";
    jq --raw-output --slurp --from-file ${fl_script} ${fl_source};
  } else {
    echo "***EXTENSION NOT IMPLEMENTED: ${ext_script}";
  } fi

  # declare -p fl_script dir_script ext_script fl_source;

  return 0;
}

function get_formats(){

    local -n ref_labels="${1}";
    local -n ref_hdr="${2}";
    local -n ref_rcd="${3}";
    local -n ref_jq="${4}";
    local schema="${5}";
    local delimiter='|';

    ref_labels='';
    ref_hdr='';
    ref_rcd='';
    ref_jq='[';

    local title field type width prec
    while IFS='|' read -r title field type width prec
    do
        # strip spaces
        # title=${title// /};
        field=${field// /};
        type=${type// /};
        width=${width// /};
        prec=${prec// /};

        title=${title//\"/};
        ref_labels+="${title} ";

        ref_hdr+="%-${width}s${delimiter}"

        case "${type}" in

            s)
                ref_rcd+="%-${width}s${delimiter}"
                ;;

            d)
                ref_rcd+="%${width}d${delimiter}"
                ;;

            f)
                ref_rcd+="%${width}.${prec}f${delimiter}"
                ;;

            *)
                ref_rcd+="%s${delimiter}"
                ;;

        esac

        ref_jq+=".${field}, "

    done <<< "${schema}"

    ref_labels=${ref_labels}-SPC-;
    ref_labels=${ref_labels/ -SPC-/};
    ref_hdr+=$'\n'
    ref_rcd+=$'\n'
    ref_jq+=']| @tsv'
    ref_jq=${ref_jq/', ]|'/'] |'}

    return 0
}

    local factor="${3:?*** MISSING FACTOR! ***}";

    awk -F'|' -v OFS='|' \
        -v sym="${symbol}" \
        -v fld="${field}" \
        -v fac="${factor}" '
    BEGIN {
        all = (sym == "*" || sym == "." || sym == "")
    }

    {
        if (all || $1 == sym) {
            $fld = sprintf("%10.3f", $fld * fac)
        }
        print
    }'
}

{

  shift 2;

  local table=$(
  {
    printf "${fmt_hdr}" ${labels};
    render_lots_view \
    | grep "$@" \
    | sort \
      --field-separator='|' \
      --key=${sort_key} \
    ;
  } \
  );

  echo "${table}" \
  | cut --delimiter='|' --fields=${columns} \
  | nl --number-width=4;

  local total=$(
    echo "${table}" \
    | grep --invert-match -e OPEN_DATE \
    | cut --delimiter='|' --fields=4 \
    | awk "${awk_total}" \
    ;
  );

  printf '%-18s: $%10.3f\n' "${report_label}" ${total};

  return 0;
}

    --key=${sort_by_dts} \
    | grep -e "^${symbol}[ |]" \
    ;
  } \
  | nl --number-width=4 \
  ;
  # | cat -n \
  return 0;
}
alias vl.sym=show_lots_by_symbol;

#-----------------------------------------------------------------------
function list_lot_symbols()
{
  require_input_data || return 1;

  echo "## List of symbols:"
  render_lots_view \
  | cut --delimiter='|' --fields=1 \
  | sort \
  | uniq \
  | awk '
  { 
    printf "%-8s", $1; 
    if (NR % 10 == 0) print "";
  }
  END {
    print "";
  }
  ' \
  | nl --number-width=4 \
  ;
  echo;

  return 0;
}
alias vl.ls=list_lot_symbols;

#-----------------------------------------------------------------------
function show_harvestable_inventory()
{
  require_input_data || return 1;

  local fl_view_lots=$(mktemp);

  render_lots_view > "${fl_view_lots}" || {
    rm -f "${fl_view_lots}";
    return 1;
  }

  while read -r sym; do
    {
      printf "${fmt_hdr}" ${labels};

      grep -e "^${sym}[ |]" "${fl_view_lots}" \
      | sort \
      --field-separator='|' \
      --key=${sort_by_dts} \
    ;
    } \
    | cut --delimiter='|' --fields=1-5,7- \
    | nl --number-width=4;

    echo -e "<==\n\n";

  done < <(
    grep --regexp='10.HVT' "${fl_view_lots}" \
    | cut --delimiter='|' --fields=1 \
    | sort \
    | uniq
  );

  rm -f "${fl_view_lots}";

  return 0;
}
alias vl.hvti=show_harvestable_inventory;

#-----------------------------------------------------------------------
function write_ib_lot_symbols()
{
  local state=${1:-HVT}
  render_lots_view                     \
  | grep ${state}                      \
  | cut -d "|" -f1                     \
  | sort                               \
  | uniq                               \
  | tr -d " "                          \
  | ib.lsd                             \
  ;
  return 0;
}

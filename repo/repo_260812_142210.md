            + ($locker.quantity | tostring)
            + "▶"
            + ($locker.open_date | sub("^20"; ""))

          .

        end

      else

        .

      end

| map(

    (
      map(
        select(
          .record_type == "lot"
          and .lock_state == "50.LOS"
        )
      )
    ) as $los

    |

    if ($los | length) == 0
    then
      .
    else

      (
        $los
        | max_by(.cost_basis_price)
        | .cost_basis_price
      ) as $max_cost

      |

      map(

        if .record_type == "lot"
        and .lock_state == "50.LOS"
        and .cost_basis_price == $max_cost
        and .cost_basis_price >= .mark_price

        then

          .lock_state = "20.PBA"

        else

          .

        end

      )

    end

)

  .originating_transaction_id,
  .originating_order_id

| map(

    if length == 1 then .[0]

      .[0] + {

          (map(.quantity) | add),

        position_value:
          (map(.position_value) | add),

        fifo_pnl_unrealized:
          (map(.fifo_pnl_unrealized) | add),

        cost_basis_price:
          (
            (map(.cost_basis_price * .quantity) | add)
            /
            (map(.quantity) | add)
          ),

        open_price:
          (
            (map(.open_price * .quantity) | add)
            /
            (map(.quantity) | add)
          )

      }

    end

)

#-----------------------------------------------------------------------
# update_mark_prices.jq
#
# Input:
#   One JSONL stream containing:
#     - Flex records (lots + summaries)
#     - Live records (record_type == "live")
#
# Output:
#   Updated Flex records only.
#-----------------------------------------------------------------------

# Build an index of live records.
reduce inputs as $r
(
  {
    "live": {},
    "flex": []
  };

  if $r.record_type == "live" then
    .live[$r.symbol] = $r
  else
    .flex += [$r]
  end
)

# Emit the updated Flex records.
| . as $db
| $db.flex[]
| if $db.live[.symbol] then
    .mark_price = $db.live[.symbol].mark_price
    | .position_value =
        (
            .mark_price
            * .quantity
            * .multiplier
        )
    | .fifo_pnl_unrealized =
        (
            .position_value
            - (
                .cost_basis_price
                * .quantity
                * .multiplier
            )
        )
  else
    .
  end

#-----------------------------------------------------------------------
define prefix_filename =
${strip $()
	${eval
		path   := ${strip $(1)}
		prefix := ${strip $(2)}
	}
	${eval
		filename  := ${notdir $(path)}
	}
	$(dir_tidata)/$(prefix)$(filename:jq=jsonl)
}
endef

#-----------------------------------------------------------------------
define _jq_arg_validation =
${eval 
	options  := ${strip $(1)}
	script   := ${strip $(2)}
	inputs   := ${strip $(3)}
	prefix   := ${strip $(4)}
}
${eval
	input1 := ${word 1, $(inputs)}
}
${eval
	fl_out  = ${dir $(input1)}$(prefix)${notdir $(input1)}
}
: echo -n                            \
	"${notdir $(input1)}"              \
	"--> ${notdir $(script)}"          \
	"--> ${notdir $(fl_out)}"          \
;
endef
#.......................................................................
define _jq_transform =
	jq $(options) --from-file 
	$(script)
	$(inputs) 
	> 
	$(input1).tmp;
endef
#.......................................................................
define _jq_publish =
	mv --force 
	$(input1).tmp 
	$(input1);
	jq .
	$(input1) 
	> 
	$(fl_out);
endef
#.......................................................................
define jq_stage  =
${call _jq_arg_validation, $(1), $(2), $(3), $(4)}
${strip $()
	${call _jq_transform}
	${call _jq_publish}
	:
}
endef

def wait_for_market_data(ib, tickers, timeout=10):
    start = time.time()

    while time.time() - start < timeout:
        if all(
            not math.isnan(t.markPrice)
            or not math.isnan(t.last)
            for t in tickers
        ):
            return True

        ib.sleep(0.20)

    return False

def get_historical_last_bar(ib: IB, contract):
    """Get last completed 1-min bar."""
    try:
        bars = ib.reqHistoricalData(
            contract,
            endDateTime='',
            durationStr='1 D',
            barSizeSetting='1 min',
            # barSizeSetting='10 secs',
            whatToShow='TRADES',
            useRTH=True,
            formatDate=1
        )
        if not bars:
            return None
        last = bars[-1]
        return last.low, last.high, (last.high + last.low) / 2, last.close
    except Exception as e:
        logging.warning(f"Historical data failed for {contract.symbol}: {e}")
        return None

def create_synthetic_ticker(contract,
                            account,
                            avg_cost,
                            bid,
                            ask,
                            mark,
                            last):
    """
    Build a ticker-like object from historical data.
    """

    class SyntheticTicker:
        def __init__(self):
            self.contract = contract
            self.account = account
            self.avgCost = avg_cost

            # Match the interface of ib_insync.Ticker
            self.marketDataType = 2
            self.time = None
            self.markPrice = mark

            self._bid = bid
            self._ask = ask
            self._last = last

        def marketPrice(self):
            return self.markPrice

        @property
        def bid(self):
            return self._bid

        @property
        def ask(self):
            return self._ask

        @property
        def last(self):
            return self._last

    return SyntheticTicker()

def is_market_open(ib):
    """
    Return True if the current IBKR server time falls within
    regular U.S. market hours (09:30–16:00 ET, Mon–Fri).
    """

    # return True  #******************** TESTING  ********************
    return False #******************** TESTING  ********************

    try:
        now = ib.reqCurrentTime()

        # Weekend
        if now.weekday() >= 5:
            return False

        minutes = now.hour * 60 + now.minute

        market_open = 4 * 60 + 30
        market_close = 20 * 60

        return market_open <= minutes < market_close

    except Exception as e:
        logging.warning(
            f"Failed to get server time: {e}. Assuming market closed."
        )
        return False

#   $1 : ref_labels

#   $4 : ref_jq
#   $4 : schema (At some point to become optional external file)

#   ${labels}

function get_formats(){

    local -n ref_labels="${1}";
    local -n ref_hdr="${2}";
    local -n ref_rcd="${3}";
    local -n ref_jq="${4}";
    local schema="${5}";
    local delimiter='|';

    ref_labels='';
    ref_hdr='';
    ref_rcd='';
    ref_jq='[';

    local title field type width prec
    while IFS='|' read -r title field type width prec
    do
        # strip spaces
        # title=${title// /};
        field=${field// /};
        type=${type// /};
        width=${width// /};
        prec=${prec// /};

        title=${title//\"/};
        ref_labels+="${title} ";

        ref_hdr+="%-${width}s${delimiter}"

        case "${type}" in

            s)
                ref_rcd+="%-${width}s${delimiter}"
                ;;

            d)
                ref_rcd+="%${width}d${delimiter}"
                ;;

            f)
                ref_rcd+="%${width}.${prec}f${delimiter}"
                ;;

            *)
                ref_rcd+="%s${delimiter}"
                ;;

        esac

        ref_jq+=".${field}, "

    done <<< "${schema}"

    ref_labels=${ref_labels}-SPC-;
    ref_labels=${ref_labels/ -SPC-/};
    ref_hdr+=$'\n'
    ref_rcd+=$'\n'
    ref_jq+=']| @tsv'
    ref_jq=${ref_jq/', ]|'/'] |'}

    return 0
}

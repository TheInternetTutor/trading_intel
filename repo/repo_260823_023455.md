# FILE: update_mark_prices.jq ::
########################################################################
#-----------------------------------------------------------------------
# update_mark_prices.jq
#
# Input:
#   One JSONL stream containing:
#     - Flex records (lots + summaries)
#     - Live records (record_type == "live")
#
# Output:
#   Updated Flex records only.
#-----------------------------------------------------------------------

# Build an index of live records.
reduce inputs as $r
(
  {
    "live": {},
    "flex": []
  };

  if $r.record_type == "live" then
    .live[$r.symbol] = $r
  else
    .flex += [$r]
  end
)

# Emit the updated Flex records.
| . as $db
| $db.flex[]
| if $db.live[.symbol] then
    .mark_price = $db.live[.symbol].mark_price
    | .position_value =
        (
            .mark_price
            * .quantity
            * .multiplier
        )
    | .fifo_pnl_unrealized =
        (
            .position_value
            - (
                .cost_basis_price
                * .quantity
                * .multiplier
            )
        )
  else
    .
  end

# FILE: consolidate_lots.jq ::
########################################################################

group_by([
  .symbol,
  .open_datetime_raw,
  .originating_transaction_id,
  .originating_order_id
])

| map(

    if length == 1 then .[0]

    else

      .[0] + {

        quantity:
          (map(.quantity) | add),

        position_value:
          (map(.position_value) | add),

        fifo_pnl_unrealized:
          (map(.fifo_pnl_unrealized) | add),

        cost_basis_price:
          (
            (map(.cost_basis_price * .quantity) | add)
            /
            (map(.quantity) | add)
          ),

        open_price:
          (
            (map(.open_price * .quantity) | add)
            /
            (map(.quantity) | add)
          )

      }

    end

)

| .[]

# FILE: classify_lkd.jq ::
########################################################################

group_by(.symbol)

| map(

    . as $lots

    |

    reduce range(0; length) as $i (

      $lots;

      if .[$i].record_type == "lot"
      and .[$i].lock_state == "10.HVT"

      and

      (
        any(
          .[$i+1:][];

          .record_type == "lot"
          and .lock_state == "50.LOS"
        )
      )

      then

        .[$i].lock_state = "30.LKD"

      else

        .

      end

    )

)

| add
| .[]
########################################################################
# FILE: classify_locking.jq ::
########################################################################
group_by(.symbol)

| map(

    . as $lots

    |

    reduce range(0; length) as $i (

      $lots;

      if .[$i].record_type == "lot"
      and .[$i].lock_state == "30.LKD"

      then

        (
          .[$i+1:]
          | map(
              select(
                .record_type == "lot"
                and .lock_state == "50.LOS"
              )
            )
          | last
        ) as $locker

        |

        if $locker then

          .[$i].locking_lot_id = $locker.lot_id

          |
          .[$i].lock_condition =
            "◆"
            + ((($locker.open_price + 3 / $locker.quantity) | floor) | tostring)
            + "x"
            + ($locker.quantity | tostring)
            + "▶"
            + ($locker.open_date | sub("^20"; ""))

        else

          .

        end

      else

        .

      end

    )

)

| add
| .[]
########################################################################
# FILE: classify_pba.jq ::
########################################################################

group_by(.symbol)

| map(

    (
      map(
        select(
          .record_type == "lot"
          and .lock_state == "50.LOS"
        )
      )
    ) as $los

    |

    if ($los | length) == 0
    then
      .
    else

      (
        $los
        | max_by(.cost_basis_price)
        | .cost_basis_price
      ) as $max_cost

      |

      map(

        if .record_type == "lot"
        and .lock_state == "50.LOS"
        and .cost_basis_price == $max_cost
        and .cost_basis_price >= .mark_price

        then

          .lock_state = "20.PBA"

        else

          .

        end

      )

    end

)

| add
| .[]
########################################################################
# FILE: aggregate_hvt.jq ::
########################################################################
. as $records

| [
    $records[]
    | select(
        .record_type == "lot"
        and .lock_state == "10.HVT"
      )
  ]

| group_by(.symbol)

| map(
    sort_by(.open_datetime_raw)
    | . as $lots
    | $lots[0] as $oldest

    | $oldest

    | .record_type = "aggregate"
    | .lock_state  = "15.HVT"

    | .quantity =
        ($lots | map(.quantity) | add)

    | .open_price =
        ($lots | map(.open_price) | max)

    | .cost_basis_price =
        ($lots | map(.cost_basis_price) | max)

    | .fifo_pnl_unrealized =
        (
            ($lots | map(.fifo_pnl_unrealized) | add)
            - 1.12
        )
  )

| ($records + .)[]
